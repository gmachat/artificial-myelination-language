CORE
app name
profile script|local|connected
record Person:
  name: Text
  scores: List[Int]
task greet(person: Person) -> Text:
  require person.name != ""
  ensure result != ""
  let message = "Hello, " + person.name
  if person.scores[0] > 0:
    return message
  else:
    fail "invalid score"
task main:
  show greet({name: "Ada", scores: [42]})

app name
profile react|web
component Counter(title: Text):
  state count: Int = 0
  view:
    column:
      heading title
      text "Count: {count}"
      when count == 0:
        text "Ready"
      else:
        text "Started"
      button "Add":
        on click:
          set count = count + 1
      input name

Build: aml build app.aml --target web -o app
